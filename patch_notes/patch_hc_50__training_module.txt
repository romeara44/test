CREATE TABLE IF NOT EXISTS `training_logs` (
  `tl_id` int(11) NOT NULL AUTO_INCREMENT,
  `tl_tlt_id` int(11) NOT NULL,
  `tl_conducted_date` date NOT NULL,
  `tl_hire_date` date NOT NULL,
  `tl_trainer` varchar(255) NOT NULL,
  `tl_attendees` text NOT NULL,
  `tl_active` tinyint(1) NOT NULL DEFAULT '1',
  `tl_archived` tinyint(1) NOT NULL DEFAULT '0',
  PRIMARY KEY (`tl_id`)
) ENGINE=InnoDB  DEFAULT CHARSET=utf8 AUTO_INCREMENT=1 ;

CREATE TABLE IF NOT EXISTS `training_log_types` (
  `tlt_id` int(11) NOT NULL AUTO_INCREMENT,
  `tlt_name` varchar(255) NOT NULL,
  PRIMARY KEY (`tlt_id`)
) ENGINE=InnoDB  DEFAULT CHARSET=utf8 AUTO_INCREMENT=1 ;

CREATE TABLE IF NOT EXISTS `security_reminders` (
  `sr_id` int(11) NOT NULL AUTO_INCREMENT,
  `sr_launched_date` date NOT NULL,
  `sr_developed_by` varchar(255) NOT NULL,
  `sr_regulation` varchar(255) CHARACTER SET utf32 NOT NULL,
  `sr_dt_id` int(11) NOT NULL,
  `sr_archived` tinyint(1) NOT NULL DEFAULT '0',
  `sr_active` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`sr_id`)
) ENGINE=InnoDB  DEFAULT CHARSET=utf8 AUTO_INCREMENT=1 ;
