  `vl_medical_record_number` varchar(50) DEFAULT NULL,
  `vl_name` varchar(255) DEFAULT NULL,
  `vl_date_of_birth` date DEFAULT NULL,
  `vl_address` varchar(255) DEFAULT NULL,
  `vl_disclosure_address` varchar(255) DEFAULT NULL,
  `vl_date_requested_from` date DEFAULT NULL,
